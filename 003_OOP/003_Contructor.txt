Ein Konstruktor wird genutzt, um einen gewissen Prozess bei der Kreierung des Objektes zu starten.

Code Beispiel
############

#include <iostream>

class Person {
public:
    int age;
	Person(int agePar);
};

Person::Person(int agePar){
    Person::age = agePar;
}

int main(){
    Person* Tom = new Person(18);
	std::cout << Tom->age;
    return 0;
}
