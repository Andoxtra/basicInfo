Eine Klasse ist ein Bauplan für ein Objekt. 
Dies nutzt man häufig um Gegebeneheiten in der reellen Welt programmtechnisch darzustellen.

Code Beispiel
############
#include <iostream>

class Person {
public:
    int age;
    void sayAge();
};

void Person::sayAge(){
    std::cout << "My age is " << Person::age;
}

int main(){
    Person* Tom = new Person();
	Tom->sayAge();
    return 0;
}
